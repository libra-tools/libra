commit = sys.argv[1]
assert re.fullmatch(r'[0-9a-f]{40}', commit), f'bad COMMIT_SHA: {commit!r}'

start = time.monotonic()

class BudgetExhausted(Exception):
    pass

def budget_left():
    left = DEADLINE - (time.monotonic() - start)  # 浮点剩余秒数，不取整、不越预算
    if left <= 0:
        raise BudgetExhausted()
    return left

LIST_FIELDS = 'databaseId,attempt,event,headBranch,headSha,status,conclusion,workflowName'  # run list 不支持 jobs 字段（实测 Unknown JSON field）
VIEW_FIELDS = 'databaseId,attempt,event,headBranch,headSha,status,conclusion,jobs,workflowName'

def gh_json(args):
    r = subprocess.run(['gh'] + REPO + args, check=True, text=True,
                       capture_output=True, timeout=budget_left())
    return json.loads(r.stdout)

def http_statuses(stderr):
    # 严格三位状态码；避免 'http 42' 之类前缀误判
    return {int(m) for m in re.findall(r'\bhttp[ :]*(\d{3})\b', stderr, re.I)}

def transient_call(error):
    # 分类顺序：严格三位 HTTP 状态优先（429、5xx、403+rate limit → 暂时；其余 4xx → 确定性）
    # 无 HTTP 状态时：若命中任何 usage/config/schema/auth 特征词 → 确定性；
    # 仅命中明确网络/连接/TLS/超时特征词或 TimeoutExpired → 暂时性；其余未知错误默认 fail-closed 为确定性（ER-10 不盲目重试）
    if isinstance(error, subprocess.TimeoutExpired):
        return True
    if not isinstance(error, subprocess.CalledProcessError):
        return False
    if error.returncode == 4:
        return False
    stderr = (error.stderr or '')
    statuses = http_statuses(stderr)
    lowered = stderr.lower()
    if statuses:
        rate_limit = 403 in statuses and 'rate limit' in lowered
        if any(code == 429 or code >= 500 or rate_limit for code in statuses):
            return True
        return False  # 其余三位状态（401/403 非 rate limit/404/409…）→ 确定性
    deterministic_words = (
        'unknown json field', 'unknown field', 'invalid field', 'unknown flag', 'unknown command',
        'unknown argument', 'invalid argument', 'flag needs an argument', 'accepts ', 'requires ',
        'could not find any workflows', 'no workflows named', 'failed to read configuration',
        'configuration', 'not logged in', 'authentication', 'permission denied', 'not a git repository'
    )
    if any(word in lowered for word in deterministic_words):
        return False  # 参数/schema/配置/权限错误 → 确定性（ER-10 不重试）
    if any(word in lowered for word in ('connection', 'network', 'tls', 'timed out', 'timeout', 'reset by peer', 'dial tcp', 'eof', 'temporary failure', 'service unavailable', 'bad gateway', 'gateway timeout', 'rate limit')):
        return True
    return False  # 无明确暂时性特征的 CalledProcessError 一律按确定性处理

class StructuralError(Exception):
    pass

REQUIRED_LIST_FIELDS = ('databaseId', 'attempt', 'event', 'headBranch', 'headSha', 'status', 'conclusion', 'workflowName')
REQUIRED_RUN_FIELDS = ('databaseId', 'attempt', 'event', 'headBranch', 'headSha', 'status', 'conclusion', 'jobs', 'workflowName')

def validate_list_runs(runs):
    if not isinstance(runs, list):
        raise StructuralError('run list payload is not a list')
    for idx, item in enumerate(runs):
        if not isinstance(item, dict):
            raise StructuralError(f'run list item {idx} is not an object')
        missing = [k for k in REQUIRED_LIST_FIELDS if k not in item]
        if missing:
            raise StructuralError(f'missing fields {missing} in run list item {idx}')
        if not isinstance(item['databaseId'], int) or isinstance(item['databaseId'], bool) or item['databaseId'] <= 0:
            raise StructuralError(f'invalid databaseId in run list item {idx}')
        if not isinstance(item['attempt'], int) or isinstance(item['attempt'], bool) or item['attempt'] < 1:
            raise StructuralError(f'invalid attempt in run list item {idx}')
        for str_key in ('event', 'headBranch', 'headSha', 'status', 'workflowName'):
            if not isinstance(item[str_key], str) or not item[str_key]:
                raise StructuralError(f'invalid {str_key} in run list item {idx}')
        if item['conclusion'] is not None and not isinstance(item['conclusion'], str):
            raise StructuralError(f'invalid conclusion in run list item {idx}')
    return runs

def validate_response(detail):
    if not isinstance(detail, dict):
        raise StructuralError('run view payload is not an object')
    missing = [k for k in REQUIRED_RUN_FIELDS if k not in detail]
    if missing:
        raise StructuralError(f'missing fields {missing} in run payload')
    if not isinstance(detail['databaseId'], int) or isinstance(detail['databaseId'], bool) or detail['databaseId'] <= 0:
        raise StructuralError('invalid databaseId in run payload')
    if not isinstance(detail['attempt'], int) or isinstance(detail['attempt'], bool) or detail['attempt'] < 1:
        raise StructuralError('invalid attempt in run payload')
    for str_key in ('event', 'headBranch', 'headSha', 'status', 'workflowName'):
        if not isinstance(detail[str_key], str) or not detail[str_key]:
            raise StructuralError(f'invalid {str_key} in run payload')
    if detail['conclusion'] is not None and not isinstance(detail['conclusion'], str):
        raise StructuralError('invalid conclusion in run payload')
    if not isinstance(detail['jobs'], list):
        raise StructuralError('jobs is not a list')
    for idx, job in enumerate(detail['jobs']):
        if not isinstance(job, dict):
            raise StructuralError(f'job {idx} is not an object')
        if 'name' not in job or not isinstance(job['name'], str) or not job['name']:
            raise StructuralError(f'job {idx} has invalid name')
        if 'conclusion' not in job or (job['conclusion'] is not None and not isinstance(job['conclusion'], str)):
            raise StructuralError(f'job {idx} has invalid conclusion')
        job_id = job.get('databaseId')
        if job_id is not None and (not isinstance(job_id, int) or isinstance(job_id, bool) or job_id <= 0):
            raise StructuralError(f'job {idx} has invalid databaseId')
    return detail

def one_run(workflow):
    runs = validate_list_runs(gh_json(['run', 'list', '--workflow', workflow, '--branch', 'main', '--event', 'push',
                                       '--commit', commit, '--limit', '20', '--json', LIST_FIELDS]))
    runs = [run for run in runs if run['event'] == 'push' and run['headBranch'] == 'main' and run['headSha'] == commit]
    if not runs:
        return None
    run = max(runs, key=lambda run: run['attempt'])
    detail = gh_json(['run', 'view', str(run['databaseId']), '--attempt', str(run['attempt']), '--json', VIEW_FIELDS])
    validate_response(detail)
    if detail['databaseId'] != run['databaseId'] or detail['attempt'] != run['attempt'] or detail['headSha'] != commit:
        raise StructuralError(f"binding mismatch: id {detail['databaseId']} vs {run['databaseId']}; attempt {detail['attempt']} vs {run['attempt']}; sha {detail['headSha'][:8]}")
    if detail['event'] != 'push' or detail['headBranch'] != 'main' or detail['headSha'] != commit:
        raise StructuralError(f"non-main/push or non-matching head: {detail['event']}/{detail['headBranch']}/{detail['headSha'][:8]}")
    return detail

def classify(detail):
    # 取消只针对仍在运行的 jobs，不能抹去已结束的失败：先看 jobs，再判纯取消
    failed_jobs = [job for job in detail['jobs'] if job.get('conclusion') in ('failure', 'timed_out', 'startup_failure')]
    if failed_jobs:
        return 'failed'
    if detail['status'] != 'completed':
        return 'running'
    if detail['conclusion'] == 'cancelled':
        return 'cancelled'
    if detail['conclusion'] != 'success' or not detail['jobs']:
        return 'failed'
    if any(job['conclusion'] != 'success' for job in detail['jobs']):
        return 'failed'
    return 'passed'

def receipt(detail_or_none, workflow):
    if detail_or_none is None:
        return f'{workflow}: not observed'
    d = detail_or_none
    jobs = ','.join(f"{job.get('databaseId') or 'no-id'}:{job.get('name')}:{job.get('conclusion')}" for job in d['jobs'])
    return f"{workflow}: {d['workflowName']} run={d['databaseId']} attempt={d['attempt']} sha={d['headSha']} status={d['status']} conclusion={d['conclusion']} jobs=[{jobs}]"

observed = {}  # 跨轮保留最近观察，所有出口都对两个 workflow 输出可用收据
query_errors = {'base.yml': 0, 'codeql.yml': 0}

def leave(code, note):
    # 统一出口：失败优先裁决已在调用前完成；这里只带收据退出
    both = [receipt(observed.get(w), w) for w in ('base.yml', 'codeql.yml')]
    for line in both:
        print(line)
    print(f'{note} for {commit}')
    sys.exit(code)

def sleep_or_exit(preferred):
    try:
        time.sleep(min(preferred, budget_left()))
    except BudgetExhausted:
        leave(2, 'PENDING B-5b deadline reached; keep remote-pending')

while True:
    states = {}
    errored = []
    budget_hit = False
    for workflow in ('base.yml', 'codeql.yml'):  # 两个 workflow 本轮都观察后再裁决，cancelled 不遮蔽失败
        try:
            detail = one_run(workflow)
        except (subprocess.TimeoutExpired, subprocess.CalledProcessError, json.JSONDecodeError, StructuralError, OSError, UnicodeDecodeError) as error:
            if transient_call(error):
                query_errors[workflow] += 1
                errored.append((workflow, error, query_errors[workflow]))
            else:
                if isinstance(error, subprocess.CalledProcessError) and 'unknown json field' in ((error.stderr or '').lower()):
                    leave(1, f'SCHEMA-ERROR {workflow}: query field rejected; do not retry: {error}')
                leave(1, f'FAIL {workflow}: deterministic query error; do not retry: {error}')
            continue
        except BudgetExhausted:
            budget_hit = True  # 预算耗尽可捕获，不遮蔽本轮已观察到的其它终态；跳过并聚合
            break
        query_errors[workflow] = 0
        if detail is not None:
            observed[workflow] = detail
            states[workflow] = classify(detail)
            if states[workflow] == 'running':
                print('PENDING', receipt(detail, workflow))
        else:
            print('PENDING', workflow, 'no run yet')
    if any(state == 'failed' for state in states.values()):  # 终态失败优先于 cancelled/pending/预算中断/超限
        leave(1, f"FAIL {[w for w, s in states.items() if s == 'failed']}: roll forward")
    # 禁止空洞真值：states 为空（如预算耗尽的最后轮）时 all() 会为真，必须显式要求两个 workflow 都已判 passed
    if states.get('base.yml') == 'passed' and states.get('codeql.yml') == 'passed':
        leave(0, f'OK B-5b all applicable main push runs success')
    if any(state == 'cancelled' for state in states.values()):
        # 无可靠跨写者互斥的 rerun 机制：保持 remote-pending，不 rerun；收据由例行同步登记
        leave(2, "CANCELLED " + ','.join(w for w, s in states.items() if s == 'cancelled') + "; no rerun without writer mutex; record via plan-status routine sync")
    if budget_hit:
        leave(2, 'PENDING B-5b deadline reached; keep remote-pending (blocked until next observation window; escalate to plan owner)')
    if errored:

        if capped:
            leave(2, f"PENDING B-5b query errors exceeded cap for {capped}; keep remote-pending (blocked; escalate to plan owner)")
        try:
            left = budget_left()
        except BudgetExhausted:
            leave(2, 'PENDING B-5b deadline reached; keep remote-pending')
        for workflow, error, n in errored:
            print(f'PENDING {workflow}: transient query error ({n}/{MAX_QUERY_RETRIES}): {error}')
        sleep_or_exit(30 * (2 ** (max(n for _, _, n in errored) - 1)))
        continue
    sleep_or_exit(INTERVAL)

退出码表：`0`=PASS（仅此码允许 main D 计入完成判据）、`1`=FAIL（终态失败——含 cancelled 但 jobs 已结束为 failure/timed_out/startup_failure 的组合——确定性查询错误、非法参数/结构或绑定校验失败，前滚修复）、`2`=PENDING（无 run、进行中、纯取消、查询错误超限、截止的如实登记；同时保持 `remote-pending`，本次交付按 ER-10 视为 blocked：由推送操作者在下一次 `plan-status.md` 例行同步登记脱敏收据并升级给计划 owner；无 run、进行中、查询错误超限或截止时，恢复条件是下一次观察窗口重新运行 B-5b（只读、可重复）；纯取消的解除条件见下方「取消与 rerun 政策」：原取消收据本身不构成恢复，仅互斥机制 ADR 落地后的新观察窗口可解除）。错误分类与仓库 `cancel_superseded_main_runs.py` 同法并收紧为显式暂时性白名单：严格三位 HTTP 状态优先——429、5xx、403+rate limit 为暂时性，其余 4xx 立即 FAIL；无 HTTP 状态时，任何 usage/config/schema/auth 特征词与 `returncode 4` 立即 FAIL，仅明确网络/连接/TLS/超时特征词或 `TimeoutExpired` 进入有界退避，其余未知 `CalledProcessError` 默认 fail-closed 为确定性 FAIL。两个 workflow 都观察后才裁决（预算耗尽为可捕获异常，不遮蔽已观察的失败）：`base=cancelled, codeql=failure` 会 exit 1。取消的 run 不是 PASS；本计划没有跨写者互斥，不执行 rerun，由 `plan-status.md` 例行同步登记取消收据；不得由其它 SHA 的 run 替代。

**取消与 rerun 政策（无 rerun 路径）：** 本计划没有跨写者的安全 rerun 机制：`base.yml` 的 main run 共享 concurrency group，重跑旧 SHA 必然可能取消其它写者较新的 main CI；`cancel-superseded-main.yml`（`.github/scripts/cancel_superseded_main_runs.py` 的 `eligible`）同样会取消旧 main SHA 上的 CodeQL run（仅排除 pull_request 系事件与 `release.yml`）。因此 B-5b 观察 run 被取消（任一 workflow）时，不执行 rerun，保持 `remote-pending`，并把取消收据（workflow/run id/attempt/head SHA/conclusion）交给下一次 `plan-status.md` 例行同步登记。若维护者未来引入仓库级互斥/发布锁（如 ER-12 的独立 ADR 落地），另立 ADR 恢复 rerun 路径；在此之前，任何绕过本政策的 rerun 视为违反本计划交付链。B-3 与终审阶段仍按正文其余条款执行。超预算/取消/超限的升级口径（ER-10）：受阻后由推送操作者在 `plan-status.md` 该例行同步内登记脱敏收据与恢复条件，并升级至计划 owner（genedna）；**解除阻塞的判据不是原有同一收据本身**——已取消的 run 依旧「可观察」，但它不能解除阻塞；恢复条件是该 SHA 出现可继续取得成功结果的同 SHA run，且只能在维护者落地互斥机制/发布锁的新 ADR 之后（此前任何 rerun 均违反本政策），B-5b 以新观察窗口取得 success；在此之前不重试、不 clock-wait 空转，也不因同一取消收据反复重启。
