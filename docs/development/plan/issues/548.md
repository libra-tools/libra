执行 nextest 前在仓库根的当前 shell 定义以下函数。先校验 cwd 是 Libra 工作区根（`Cargo.toml` 与 `.config/nextest.toml` 须同在），再逐次复制现行生成配置，仅移除默认 JUnit 输出段并保留所有 test-group/override，锁定 default profile 并拒绝 profile 覆盖；临时目录用 `mktemp -d` 仓外新建，以 0700 权限和 `open("x")` 独占写入防覆盖，退出即删除，不是证据文件。TOCTOU/多进程并发：目录名随机且仓外、0600 独占创建，攻击者无法预占或读取他人目录；CI 并行时各进程独立目录，源配置结构改变或校验失败就停止并修订计划；所有 nextest 命令均通过它运行。

```bash
libra_plan_nextest() (
  set -e
  LIBRA_PLAN_MANIFEST="$(pwd -P)/Cargo.toml"
  LIBRA_PLAN_MODE=run
  if [ "${1:-}" = run ] || [ "${1:-}" = list ]; then LIBRA_PLAN_MODE=$1; shift; fi
  for LIBRA_PLAN_ARG in "$@"; do
    case "$LIBRA_PLAN_ARG" in -P|--profile|--profile=*|-P?*) printf '%s\n' 'profile override refused' >&2; exit 1;; esac
  done
  if [ ! -f "$LIBRA_PLAN_MANIFEST" ] || [ ! -f "$(pwd -P)/.config/nextest.toml" ]; then echo 'run at Libra workspace root' >&2; exit 1; fi
  LIBRA_PLAN_RUN_DIR="$(mktemp -d /tmp/libra-plan-548.XXXXXXXX)"
  case "$LIBRA_PLAN_RUN_DIR" in /tmp/libra-plan-548.*) ;; *) exit 1;; esac
  trap 'rm -rf -- "$LIBRA_PLAN_RUN_DIR"' EXIT
  python3 -I -c 'import os,pathlib,sys; r=pathlib.Path(sys.argv[1]).resolve(strict=True); root=pathlib.Path.cwd().resolve(); assert r.parent==pathlib.Path("/tmp").resolve() and root!=r and root not in r.parents; os.chmod(r,0o700); s=(root/".config/nextest.toml").read_text(); b="[profile.default.junit]\npath = \"junit.xml\"\n"; assert s.count(b)==1; (r/"nextest.toml").open("x").write(s.replace(b,"",1))' "$LIBRA_PLAN_RUN_DIR"
  cargo nextest "$LIBRA_PLAN_MODE" --manifest-path "$LIBRA_PLAN_MANIFEST" --config-file "$LIBRA_PLAN_RUN_DIR/nextest.toml" --user-config-file none --profile default "$@"
```

```bash
    sys.exit('nextest list failed')

    sys.exit('nextest run failed')
events = collections.defaultdict(list)
for line in run.stdout.splitlines():
    event = json.loads(line)
    if event.get('type') == 'test':
```
